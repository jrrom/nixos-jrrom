(defconst my/deepseek-key-file
  (expand-file-name "deepseek-key" user-emacs-directory))

(defun my/deepseek-api-key ()
  "Return the DeepSeek API key, prompting and saving it if necessary."
  (if (file-exists-p my/deepseek-key-file)
      (string-trim
       (with-temp-buffer
         (insert-file-contents my/deepseek-key-file)
         (buffer-string)))
    (let ((key (read-passwd "DeepSeek API key: ")))
      (make-directory (file-name-directory my/deepseek-key-file) t)
      (with-temp-file my/deepseek-key-file
        (insert key))
      (set-file-modes my/deepseek-key-file #o600)
      key)))

  (let ((base 1.0))
    (set-face-attribute 'org-level-1 nil :height 1.4 :weight 'bold)
    (set-face-attribute 'org-level-2 nil :height 1.25 :weight 'bold)
    (set-face-attribute 'org-level-3 nil :height 1.15 :weight 'bold)
    (set-face-attribute 'org-level-4 nil :height 1.05 :weight 'bold)
